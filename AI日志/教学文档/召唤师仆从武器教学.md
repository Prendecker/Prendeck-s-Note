# 召唤师仆从武器完整教学

> 适用版本：tModLoader 1.4.4 稳定版

---

## 一、仆从系统的三个零件

一个完整的召唤师仆从武器需要 **三个文件**：

```
1. ModBuff        → 仆从 Buff（让仆从活着 + 允许取消）
2. ModItem        → 召唤武器（点击时生成仆从）
3. ModProjectile  → 仆从弹幕（AI 逻辑 + 碰撞伤害）
```

它们的关系：
```
玩家使用武器(ModItem)
  → 生成仆从弹幕(ModProjectile)
  → 同时给玩家添加 Buff(ModBuff)
  → Buff 每帧检查：仆从还活着吗？是→刷新 Buff 时间；否→移除 Buff
```

> ⚠️ 你现有的 `Painted_Brush_BOOM` 不是仆从！
> 它是一个一次性弹幕（60 帧后自动销毁），没有 `minion = true`，没有 Buff 关联，没有追踪 AI。

## 二、ModBuff — 仆从的"生命线"

### 为什么需要 Buff？

原版泰拉的仆从机制：
- 召唤武器使用后，给玩家添加一个 Buff
- Buff 存在期间，仆从弹幕的 `timeLeft` 每帧被刷新（永远不死）
- 玩家右键 Buff 图标 → Buff 消失 → 仆从失去刷新 → 自然死亡

### 最小 Buff 模板

```csharp
using Terraria;
using Terraria.ModLoader;

namespace PrendeckOddments.Content.Buffs
{
    public class MyMinionBuff : ModBuff
    {
        public override void SetStaticDefaults()
        {
            Main.buffNoTimeDisplay[Type] = true;  // 不显示倒计时
            Main.buffNoSave[Type] = true;         // 不保存到世界文件
        }

        public override void Update(Player player, ref int buffIndex)
        {
            if (player.ownedProjectileCounts[ModContent.ProjectileType<MyMinionProjectile>()] > 0)
            {
                player.buffTime[buffIndex] = 18000; // 仆从活着→延长时间
            }
            else
            {
                player.DelBuff(buffIndex); // 仆从死了→移除 Buff
                buffIndex--; // 重要！索引减一
            }
        }
    }
}
```

### 关键点

| 属性 | 作用 |
|------|------|
| `Main.buffNoTimeDisplay[Type] = true` | 不显示倒计时 |
| `Main.buffNoSave[Type] = true` | 退出重进不残留 |
| `player.ownedProjectileCounts[...]` | 检查仆从是否活着 |
| `buffIndex--` | DelBuff 后修正索引 |

## 三、ModItem — 召唤武器

```csharp
using Microsoft.Xna.Framework;
using Terraria;
using Terraria.DataStructures;
using Terraria.ID;
using Terraria.ModLoader;

namespace PrendeckOddments.Content.Items.Weapons.Summon
{
    public class MyMinionStaff : ModItem
    {
        public override void SetStaticDefaults()
        {
            Item.ResearchUnlockCount = 1;
        }

        public override void SetDefaults()
        {
            Item.damage = 50;
            Item.knockBack = 4f;
            Item.mana = 10;
            Item.useTime = 30;
            Item.useAnimation = 30;
            Item.useStyle = ItemUseStyleID.Swing;

            // 召唤武器专属
            Item.DamageType = DamageClass.Summon;
            Item.buffType = ModContent.BuffType<MyMinionBuff>();
            Item.shoot = ModContent.ProjectileType<MyMinionProjectile>();
            // 没有 shootSpeed！仆从有自己的 AI

            Item.width = 32;
            Item.height = 32;
            Item.UseSound = SoundID.Item44;
            Item.autoReuse = true;
            Item.value = Item.buyPrice(0, 5, 0, 0);
            Item.rare = ItemRarityID.Blue;
        }

        public override bool Shoot(Player player, EntitySource_ItemUse_WithAmmo source,
            Vector2 position, Vector2 velocity, int type, int damage, float knockback)
        {
            player.AddBuff(Item.buffType, 2); // 添加 Buff

            var projectile = Projectile.NewProjectileDirect(
                source, position, velocity, type, damage, knockback, Main.myPlayer);
            projectile.originalDamage = Item.damage;

            return false; // 阻止原版自动生成
        }

        public override void ModifyShootStats(Player player, ref Vector2 position,
            ref Vector2 velocity, ref int type, ref int damage, ref float knockback)
        {
            position = Main.MouseWorld;
        }
    }
}
```

## 四、ModProjectile — 仆从弹幕（核心！）

```csharp
using Microsoft.Xna.Framework;
using Terraria;
using Terraria.ID;
using Terraria.ModLoader;

namespace PrendeckOddments.Content.Projectiles
{
    public class MyMinionProjectile : ModProjectile
    {
        public override void SetStaticDefaults()
        {
            Main.projFrames[Projectile.type] = 4;
            ProjectileID.Sets.MinionTargettingFeature[Projectile.type] = true;
            Main.projPet[Projectile.type] = true;
            ProjectileID.Sets.MinionSacrificable[Projectile.type] = true;
            ProjectileID.Sets.CultistIsResistantTo[Projectile.type] = true;
        }

        public override void SetDefaults()
        {
            Projectile.width = 26;
            Projectile.height = 26;
            Projectile.friendly = true;
            Projectile.minion = true;
            Projectile.DamageType = DamageClass.Summon;
            Projectile.minionSlots = 1f;
            Projectile.penetrate = -1;
            Projectile.usesLocalNPCImmunity = true;
            Projectile.localNPCHitCooldown = 10;
        }

        public override bool? CanCutTiles() => false;
        public override bool MinionContactDamage() => true;

        public override void AI()
        {
            Player player = Main.player[Projectile.owner];

            // 存活检查
            if (player.dead || !player.active)
                player.ClearBuff(ModContent.BuffType<MyMinionBuff>());
            if (player.HasBuff(ModContent.BuffType<MyMinionBuff>()))
                Projectile.timeLeft = 2;

            // 寻找目标
            int attackTarget = -1;
            Projectile.Minion_FindTargetInRange(700, ref attackTarget, false);

            Vector2 targetCenter;
            bool foundTarget = attackTarget != -1;

            if (foundTarget)
            {
                targetCenter = Main.npc[attackTarget].Center;
                Projectile.friendly = true;
            }
            else
            {
                targetCenter = player.Center;
                targetCenter.Y -= 48f;
                targetCenter.X += (10 + Projectile.minionPos * 40) * -player.direction;
                Projectile.friendly = false;
            }

            // 移动（核心公式）
            float speed = 8f;
            float inertia = 40f;

            Vector2 direction = targetCenter - Projectile.Center;
            float distance = direction.Length();

            if (distance > 2000f)
            {
                Projectile.Center = player.Center;
                Projectile.velocity = Vector2.Zero;
                return;
            }

            direction.Normalize();
            direction *= speed;
            Projectile.velocity = (Projectile.velocity * (inertia - 1) + direction) / inertia;

            // 动画
            Projectile.frameCounter++;
            if (Projectile.frameCounter >= 8)
            {
                Projectile.frameCounter = 0;
                Projectile.frame++;
                if (Projectile.frame >= Main.projFrames[Projectile.type])
                    Projectile.frame = 0;
            }

            // 视觉效果
            Lighting.AddLight(Projectile.Center, Color.White.ToVector3() * 0.78f);
            Projectile.rotation = Projectile.velocity.X * 0.05f;
        }
    }
}
```

## 五、追踪 AI 核心公式

```
新速度 = (旧速度 × (惯性-1) + 目标方向 × 速度) / 惯性
```

参数含义：
- `speed`：移动速度
- `inertia`：惯性（越大转向越慢，20-40 最常用）
